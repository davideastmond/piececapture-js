import { getNeighbors } from "./board-utils";
import { BOARD_SIZE } from "./constants";
import { isPieceCaptured } from "./game-engine";
import type { CellValue, Coordinate, PlayerId } from "./types";

function getCellOwner(cell: CellValue): PlayerId | null {
  if (cell === null) return null;
  if (typeof cell === "string") return cell;
  return cell.owner;
}

function isCellCaptured(cell: CellValue): boolean {
  return typeof cell === "object" && cell !== null && cell.isCaptured;
}

function cloneBoard(board: CellValue[][]): CellValue[][] {
  return board.map((row) =>
    row.map((cell) =>
      cell === null ? null : typeof cell === "string" ? cell : { ...cell },
    ),
  );
}

function countOpponentSurrounding(
  board: CellValue[][],
  row: number,
  col: number,
  opponentId: PlayerId,
): number {
  const neighbors = getNeighbors(row, col);
  return neighbors.filter((n) => {
    const cell = board[n.row][n.col];
    return getCellOwner(cell) === opponentId && !isCellCaptured(cell);
  }).length;
}

function wouldThisMoveLeaveCpuVulnerable(
  board: CellValue[][],
  row: number,
  col: number,
  cpuId: PlayerId,
): boolean {
  const humanId: PlayerId = cpuId === "player1" ? "player2" : "player1";
  const simulatedBoard = cloneBoard(board);
  simulatedBoard[row][col] = { owner: cpuId, isCaptured: false };

  return getNeighbors(row, col).some((neighbor) => {
    if (board[neighbor.row][neighbor.col] !== null) {
      return false;
    }

    const responseBoard = cloneBoard(simulatedBoard);
    responseBoard[neighbor.row][neighbor.col] = humanId;

    return isPieceCaptured(responseBoard, row, col, cpuId);
  });
}

function getImmediateCaptureGain(
  board: CellValue[][],
  row: number,
  col: number,
  playerId: PlayerId,
): number {
  const simulatedBoard = cloneBoard(board);
  simulatedBoard[row][col] = { owner: playerId, isCaptured: false };

  let capturedCount = 0;
  getNeighbors(row, col).forEach((n) => {
    const targetCell = simulatedBoard[n.row][n.col];
    const targetOwner = getCellOwner(targetCell);
    const opponentId: PlayerId = playerId === "player1" ? "player2" : "player1";

    if (targetOwner === opponentId && !isCellCaptured(targetCell)) {
      if (isPieceCaptured(simulatedBoard, n.row, n.col, opponentId)) {
        capturedCount += 1;
      }
    }
  });

  return capturedCount;
}

/**
 * Calculates an aggressive strategic priority score for an empty cell.
 */
export function evaluateMoveScore(
  board: CellValue[][],
  row: number,
  col: number,
  cpuId: PlayerId,
): number {
  const humanId: PlayerId = cpuId === "player1" ? "player2" : "player1";
  let score = 0;

  const neighbors = getNeighbors(row, col);
  const immediateCaptureGain = getImmediateCaptureGain(board, row, col, cpuId);
  const isDangerous = wouldThisMoveLeaveCpuVulnerable(board, row, col, cpuId);

  score += immediateCaptureGain * 12000;

  if (immediateCaptureGain > 0 && !isDangerous) {
    score += 5000;
  }

  if (isDangerous) {
    score -= 8000;
  }

  // Attack the player directly when the move is safe.
  neighbors.forEach((n) => {
    const targetCell = board[n.row][n.col];
    const targetOwner = getCellOwner(targetCell);

    if (targetOwner === humanId && !isCellCaptured(targetCell)) {
      if (!wouldThisMoveLeaveCpuVulnerable(board, row, col, cpuId)) {
        score += 900;
      }
    }
  });

  // If this move would allow the human to capture one of our pieces on the next turn,
  // avoid it unless there is a major immediate reward.
  const defensiveRisk = neighbors.some((n) => {
    const adjacentCell = board[n.row][n.col];
    const adjacentOwner = getCellOwner(adjacentCell);

    if (adjacentOwner !== cpuId || isCellCaptured(adjacentCell)) {
      return false;
    }

    const validTotalBoundaries = getNeighbors(n.row, n.col).length;
    const currentHumanBlockers = countOpponentSurrounding(
      board,
      n.row,
      n.col,
      humanId,
    );

    return currentHumanBlockers === validTotalBoundaries - 1;
  });

  if (defensiveRisk) {
    score -= 6000;
  }

  // Value pressure on human pieces while staying safe.
  neighbors.forEach((n) => {
    const adjacentCell = board[n.row][n.col];
    const adjacentOwner = getCellOwner(adjacentCell);

    if (adjacentOwner === humanId && !isCellCaptured(adjacentCell)) {
      score += 150;
    }
  });

  const simulatedBoard = cloneBoard(board);
  simulatedBoard[row][col] = { owner: cpuId, isCaptured: false };

  if (isPieceCaptured(simulatedBoard, row, col, cpuId)) {
    score -= 10000;
  }

  const isRowEdge = row === 0 || row === BOARD_SIZE - 1;
  const isColEdge = col === 0 || col === BOARD_SIZE - 1;

  if (isRowEdge && isColEdge) score += 20;
  else if (isRowEdge || isColEdge) score += 8;
  else score += 2;

  return score;
}

/**
 * Scans the board and isolates the single highest priority move.
 */
export function getBestCPUMove(
  board: CellValue[][],
  cpuId: PlayerId,
): Coordinate | null {
  let bestScore = -Infinity;
  const candidates: Coordinate[] = [];

  for (let row = 0; row < BOARD_SIZE; row++) {
    for (let col = 0; col < BOARD_SIZE; col++) {
      if (board[row][col] === null) {
        const currentScore = evaluateMoveScore(board, row, col, cpuId);

        if (currentScore > bestScore) {
          bestScore = currentScore;
          candidates.length = 0; // Clear lower-scoring ties
          candidates.push({ row, col });
        } else if (currentScore === bestScore) {
          candidates.push({ row, col });
        }
      }
    }
  }

  if (candidates.length > 0) {
    // Break ties randomly so gameplay feels organic
    const randomIndex = Math.floor(Math.random() * candidates.length);
    return candidates[randomIndex];
  }

  return null;
}
